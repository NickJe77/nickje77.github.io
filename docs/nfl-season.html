import json import os import time import requests import pandas as pd # Adjust if your NFL JSON folder is different NFL_FOLDER = "docs/data/nfl" SEASON_FILE = os.path.join(NFL_FOLDER, "nfl_2025.json") # Super Bowl LX (for 2025 season) boxscore SB_URL = "https://www.pro-football-reference.com/boxscores/202602080nwe.htm" SB_GAME_ID = "202602080nwe" # matches PFR slug (date + home team) def fetch_linescore(url: str): """ Pull the linescore (quarter by quarter + final) from PFR using pandas. Returns a list of two dicts matching your JSON style keys: "Unnamed: 0", "Unnamed: 1", "1","2","3","4","Final" """ html = requests.get(url, timeout=20).text # PFR linescore table is consistently parseable via read_html tables = pd.read_html(html) # Find the linescore table by column names linescore_df = None for df in tables: cols = [str(c) for c in df.columns] if "Final" in cols and any(c in cols for c in ["1", "Q1"]): # Prefer the one that has quarter columns 1-4 and Final and a Team column if "1" in cols and "2" in cols and "3" in cols and "4" in cols and "Final" in cols: linescore_df = df break if linescore_df is None: raise RuntimeError("Could not find linescore table on PFR page.") # Determine team name column (PFR uses "Team" in many tables) team_col = None for c in linescore_df.columns: if str(c).lower() in ["team", "tm"]: team_col = c break if team_col is None: # fallback: first column often is team team_col = linescore_df.columns[0] # Normalize to exactly two rows linescore_df = linescore_df.iloc[0:2].copy() # Build in your schema rows = [] for _, r in linescore_df.iterrows(): team_name = str(r[team_col]).strip() row = { "Unnamed: 0": "", "Unnamed: 1": team_name, "1": int(r["1"]) if pd.notna(r["1"]) else 0, "2": int(r["2"]) if pd.notna(r["2"]) else 0, "3": int(r["3"]) if pd.notna(r["3"]) else 0, "4": int(r["4"]) if pd.notna(r["4"]) else 0, "Final": int(r["Final"]) if pd.notna(r["Final"]) else 0 } rows.append(row) return rows def patch_season(): if not os.path.exists(SEASON_FILE): raise FileNotFoundError(f"Season file not found: {SEASON_FILE}") with open(SEASON_FILE, "r", encoding="utf-8") as f: data = json.load(f) if "games" not in data or not isinstance(data["games"], list): raise RuntimeError("Season JSON does not have a 'games' list.") # Check if SB already exists for g in data["games"]: if g.get("game_id") == SB_GAME_ID or (g.get("url") == SB_URL): print("Super Bowl already present in nfl_2025.json — no changes made.") return print("Fetching Super Bowl linescore from PFR...") score_rows = fetch_linescore(SB_URL) new_game = { "game_id": SB_GAME_ID, "url": SB_URL, "score": score_rows # We are intentionally NOT adding team_stats etc here. # Your season page only needs score to display. # If you later want full play-by-play tables, we can extend this. } data["games"].append(new_game) # Keep file tidy: sort by game_id data["games"].sort(key=lambda x: str(x.get("game_id", ""))) with open(SEASON_FILE, "w", encoding="utf-8") as f: json.dump(data, f, indent=2) print("Done. Super Bowl appended to:", SEASON_FILE) if __name__ == "__main__": patch_season()
