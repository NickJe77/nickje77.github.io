/* All Star Mile - Winners - Predictive search via datalist - Year filter - Robust key mapping (handles different JSON column names) */ (() => { // ✅ Set your data file here // If your JSON is elsewhere, ONLY change this line. const DATA_URL = "./data/all-star-mile.json"; const els = { yearSelect: document.getElementById("yearSelect"), queryInput: document.getElementById("queryInput"), nameSuggestions: document.getElementById("nameSuggestions"), clearBtn: document.getElementById("clearBtn"), status: document.getElementById("status"), tbody: document.getElementById("tbody"), note: document.getElementById("note"), }; let allRows = []; let winners = []; // ---------- helpers ---------- const norm = (v) => (v ?? "").toString().trim(); const lower = (v) => norm(v).toLowerCase(); const pick = (obj, keys) => { for (const k of keys) { if (obj && Object.prototype.hasOwnProperty.call(obj, k) && obj[k] !== null && obj[k] !== undefined && norm(obj[k]) !== "") { return obj[k]; } } return ""; }; const toYear = (obj) => { const y = pick(obj, ["year", "Year", "YEAR"]); if (y) return parseInt(y, 10); const d = pick(obj, ["date", "Date", "DATE"]); // try dd/mm/yyyy or yyyy-mm-dd etc const s = norm(d); const m1 = s.match(/(\d{4})/); return m1 ? parseInt(m1[1], 10) : ""; }; const toDistance = (obj) => pick(obj, ["distance", "Distance", "DISTANCE", "dist", "Dist"]); const toTrack = (obj) => pick(obj, ["track", "Track", "TRACK", "venue", "Venue"]); const toDate = (obj) => pick(obj, ["date", "Date", "DATE"]); const toRace = (obj) => pick(obj, ["race", "Race", "RACE"]); const toHorse = (obj) => pick(obj, ["horse", "Horse", "HORSE", "winner", "Winner", "WINNER"]); const toJockey = (obj) => pick(obj, ["jockey", "Jockey", "JOCKEY"]); const toTrainer = (obj) => pick(obj, ["trainer", "Trainer", "TRAINER"]); const toPos = (obj) => pick(obj, ["pos", "Pos", "position", "Position", "placing", "Placing", "finish", "Finish"]); const isWinnerRow = (obj) => { // Many datasets either: one row per runner with a pos/placing // OR only winners list. const p = norm(toPos(obj)); if (!p) { // If no position exists, assume it's a winners-only file return true; } const pl = lower(p); return pl === "1" || pl === "1st" || pl === "first" || pl === "winner" || pl === "win"; }; const uniqueSorted = (arr) => Array.from(new Set(arr)).sort((a, b) => { // numeric sort if both look like years const na = Number(a), nb = Number(b); if (!Number.isNaN(na) && !Number.isNaN(nb)) return na - nb; return String(a).localeCompare(String(b)); }); const setStatus = (msg, count) => { if (typeof count === "number") { els.status.innerHTML = `${msg} ${count}`; } else { els.status.textContent = msg; } }; // ---------- render ---------- const buildYearDropdown = () => { const years = uniqueSorted(winners.map(r => r.year).filter(Boolean)).reverse(); els.yearSelect.innerHTML = ""; const optAll = document.createElement("option"); optAll.value = "ALL"; optAll.textContent = "All Years"; els.yearSelect.appendChild(optAll); for (const y of years) { const opt = document.createElement("option"); opt.value = String(y); opt.textContent = String(y); els.yearSelect.appendChild(opt); } }; const buildSuggestions = () => { // Predictive: use unique names across winner rows (horse, jockey, trainer) const names = []; for (const r of winners) { if (r.horse) names.push(r.horse); if (r.jockey) names.push(r.jockey); if (r.trainer) names.push(r.trainer); } const list = uniqueSorted(names.filter(Boolean)); els.nameSuggestions.innerHTML = ""; for (const n of list) { const opt = document.createElement("option"); opt.value = n; els.nameSuggestions.appendChild(opt); } }; const renderTable = (rows) => { els.tbody.innerHTML = ""; const frag = document.createDocumentFragment(); for (const r of rows) { const tr = document.createElement("tr"); tr.innerHTML = ` ${r.year ?? ""} ${r.horse ?? ""} ${r.jockey ?? ""} ${r.trainer ?? ""} ${r.distance ?? ""} ${r.track ?? ""} ${r.date ?? ""} `; frag.appendChild(tr); } els.tbody.appendChild(frag); setStatus("Showing", rows.length); }; const applyFilters = () => { const yearVal = els.yearSelect.value; const q = lower(els.queryInput.value); let rows = winners.slice(); if (yearVal && yearVal !== "ALL") { const y = parseInt(yearVal, 10); rows = rows.filter(r => r.year === y); } if (q) { rows = rows.filter(r => { return lower(r.horse).includes(q) || lower(r.jockey).includes(q) || lower(r.trainer).includes(q); }); } renderTable(rows); const parts = []; if (yearVal && yearVal !== "ALL") parts.push(`Year: ${yearVal}`); if (q) parts.push(`Search: "${els.queryInput.value.trim()}"`); els.note.textContent = parts.length ? parts.join(" • ") : ""; }; const clearAll = () => { els.yearSelect.value = "ALL"; els.queryInput.value = ""; els.note.textContent = ""; applyFilters(); }; // ---------- load ---------- const load = async () => { setStatus("Loading…"); try { const res = await fetch(DATA_URL, { cache: "no-store" }); if (!res.ok) throw new Error(`Fetch failed: ${res.status} ${res.statusText}`); const data = await res.json(); if (!Array.isArray(data)) throw new Error("JSON is not an array of records."); allRows = data; winners = allRows .filter(isWinnerRow) .map(obj => ({ year: toYear(obj), horse: norm(toHorse(obj)), jockey: norm(toJockey(obj)), trainer: norm(toTrainer(obj)), distance: norm(toDistance(obj)), track: norm(toTrack(obj)), date: norm(toDate(obj)), race: norm(toRace(obj)), })) .filter(r => r.horse || r.jockey || r.trainer) .sort((a, b) => (b.year || 0) - (a.year || 0)); buildYearDropdown(); buildSuggestions(); applyFilters(); setStatus("Loaded winners:", winners.length); } catch (err) { console.error(err); setStatus("Error loading data"); els.note.textContent = `Check DATA_URL path. Current: ${DATA_URL}`; els.tbody.innerHTML = `Could not load JSON. Open DevTools Console for details.`; } }; // ---------- events ---------- els.yearSelect.addEventListener("change", applyFilters); els.queryInput.addEventListener("input", applyFilters); els.clearBtn.addEventListener("click", clearAll); load(); })();
